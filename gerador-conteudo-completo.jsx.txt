const FAKE_USERS = {
  "demo@email.com": { password: "123456", name: "Usuário Demo", credits: 5, plan: "gratis" },
};

const PLANS = {
  gratis:   { label: "Grátis",   color: "#64748b", credits: 5,   price: "R$ 0" },
  basico:   { label: "Básico",   color: "#10b981", credits: 30,  price: "R$ 97/mês" },
  pro:      { label: "Pro",      color: "#6366f1", credits: 999, price: "R$ 197/mês" },
};

const CONTENT_TYPES = [
  { id: "instagram", label: "Instagram",    icon: "📸", desc: "Legenda + hashtags" },
  { id: "email",     label: "Email Mkt",    icon: "📧", desc: "Email de vendas" },

// ============================================================
// COMPONENTE PRINCIPAL
// ============================================================
export default function App() {
  const [screen, setScreen] = useState("login"); // login | register | app | upgrade
  const [user, setUser] = useState(null);
  const [users, setUsers] = useState(FAKE_USERS);

  const login = (email, password) => {
    const u = users[email];
    if (u && u.password === password) {
      setUser({ ...u, email });
      setScreen("app");
      return null;
    }
    return "Email ou senha incorretos.";
  };

  const register = (name, email, password) => {
    if (users[email]) return "Este email já está cadastrado.";
    const newUser = { password, name, credits: 5, plan: "gratis" };
    setUsers(prev => ({ ...prev, [email]: newUser }));
    setUser({ ...newUser, email });
    setScreen("app");
    return null;
  };

  const useCredit = () => {
    setUser(prev => ({ ...prev, credits: prev.credits - 1 }));
  };

  if (screen === "login")   return <LoginScreen onLogin={login} onGoRegister={() => setScreen("register")} />;
  if (screen === "register") return <RegisterScreen onRegister={register} onGoLogin={() => setScreen("login")} />;
  if (screen === "upgrade") return <UpgradeScreen user={user} onBack={() => setScreen("app")} />;
  return <GeneratorScreen user={user} onUpgrade={() => setScreen("upgrade")} onLogout={() => { setUser(null); setScreen("login"); }} useCredit={useCredit} />;
}

// ============================================================
// TELA DE LOGIN
// ============================================================
function LoginScreen({ onLogin, onGoRegister }) {
  const [email, setEmail] = useState("demo@email.com");
  const [password, setPassword] = useState("123456");
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);

  const handle = async () => {
    setLoading(true);
    await new Promise(r => setTimeout(r, 600));
    const err = onLogin(email, password);
    if (err) setError(err);
    setLoading(false);
  };

  return (
    <Bg>
      <Card>

        <h2 style={styles.cardTitle}>Entrar na sua conta</h2>

        <div style={styles.hint}>💡 Teste rápido: email <b>demo@email.com</b> / senha <b>123456</b></div>
        <Input label="Email" value={email} onChange={setEmail} placeholder="seu@email.com" />
        <Input label="Senha" value={password} onChange={setPassword} placeholder="••••••" type="password" />
        {error && <div style={styles.error}>{error}</div>}
        <Btn onClick={handle} loading={loading}>Entrar</Btn>
        <p style={styles.switchText}>Não tem conta? <span style={styles.link} onClick={onGoRegister}>Criar grátis</span></p>

// ============================================================
// TELA DE CADASTRO
// ============================================================
function RegisterScreen({ onRegister, onGoLogin }) {
  const [name, setName] = useState("");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);

  const handle = async () => {
    if (!name || !email || !password) { setError("Preencha todos os campos."); return; }
    setLoading(true);
    await new Promise(r => setTimeout(r, 600));
    const err = onRegister(name, email, password);
    if (err) setError(err);
    setLoading(false);
  };

  return (
    <Bg>
      <Card>
        <Logo />
        <h2 style={styles.cardTitle}>Criar conta grátis</h2>
        <p style={styles.cardSub}>5 gerações grátis para começar agora</p>
        <Input label="Seu nome" value={name} onChange={setName} placeholder="João Silva" />
        <Input label="Email" value={email} onChange={setEmail} placeholder="seu@email.com" />
        <Input label="Senha" value={password} onChange={setPassword} placeholder="Mínimo 6 caracteres" type="password" />
        {error && <div style={styles.error}>{error}</div>}
        <Btn onClick={handle} loading={loading}>Criar conta grátis</Btn>
        <p style={styles.switchText}>Já tem conta? <span style={styles.link} onClick={onGoLogin}>Entrar</span></p>
      </Card>
    </Bg>
  );

  const [copied, setCopied] = useState(false);

  const noCredits = user.credits <= 0;

    if (!topic.trim() || noCredits) return;
    setLoading(true);

    const systemPrompt = `Você é um copywriter expert em marketing digital para o mercado brasileiro.
Crie conteúdo de alta conversão, autêntico e natural.
Retorne APENAS o conteúdo final pronto para uso, sem explicações.`;

Tom: ${tone}
Público: ${audience || "geral"}
Canal: ${type.desc}

    try {
      const res = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: { "Content-Type": "application/json" },

          model: "claude-sonnet-4-20250514",
          max_tokens: 1000,
          system: systemPrompt,
          messages: [{ role: "user", content: userPrompt }],
        }),
      });
      const data = await res.json();
      const text = data.content?.map(b => b.text || "").join("") || "Erro ao gerar.";

      setTotalGenerated(n => n + 1);

      setResult("Erro de conexão. Tente novamente.");

    setLoading(false);

        </div>
      </div>

      <div style={{ maxWidth: 820, margin: "0 auto", padding: "32px 20px" }}>

      <input type={type} value={value} onChange={e => onChange(e.target.value)} placeholder={placeholder}
        style={{ width: "100%", background: "rgba(255,255,255,0.07)", border: "1px solid rgba(255,255,255,0.12)", borderRadius: 10, padding: "12px 14px", color: "#e8e8f0", fontSize: 15, fontFamily: "inherit", outline: "none", boxSizing: "border-box" }}
        onFocus={e => e.target.style.borderColor = "rgba(99,102,241,0.7)"}
        onBlur={e => e.target.style.borderColor = "rgba(255,255,255,0.12)"}
      />
    </div>
  );
}
